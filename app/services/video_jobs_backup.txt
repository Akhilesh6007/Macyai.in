
"""
Macy AI - Background Video Job Service

Creates video tasks, saves progress, and combines
the existing Pixabay, Edge TTS, subtitle and
FFmpeg services.
"""

import json
import logging
import uuid
import time
from datetime import datetime, timezone
from pathlib import Path


from app.services.material import (
    search_videos,
    download_video,
    MaterialError,
)
from app.services.voice import generate_voiceover_sync
from app.services.subtitle import generate_subtitles
from app.services.video import render_video


ROOT = Path(__file__).resolve().parents[2]
TASKS_DIR = ROOT / "storage" / "tasks"

logger = logging.getLogger(__name__)


def update_job_status(task_id, **changes):
    """Update task.json safely."""

    task_dir = TASKS_DIR / task_id
    task_file = task_dir / "task.json"

    if task_file.exists():
        status = json.loads(
            task_file.read_text(encoding="utf-8")
        )
    else:
        status = {"task_id": task_id}

    status.update(changes)

    status["updated_at"] = datetime.now(
        timezone.utc
    ).isoformat()

    temporary = task_dir / "task.json.tmp"

    temporary.write_text(
        json.dumps(status, indent=2),
        encoding="utf-8",
    )

    temporary.replace(task_file)

    return status


def create_video_job(subject, script, keywords):
    """Create a new task with a unique ID."""

    task_id = uuid.uuid4().hex[:12]
    task_dir = TASKS_DIR / task_id

    task_dir.mkdir(
        parents=True,
        exist_ok=False,
    )

    manifest = {
        "subject": subject,
        "script": script,
        "keywords": keywords,
        "videos": [],
    }

    (task_dir / "manifest.json").write_text(
        json.dumps(manifest, indent=2),
        encoding="utf-8",
    )

    update_job_status(
        task_id,
        state="queued",
        stage="queued",
        subject=subject,
        output=None,
        error=None,
    )

    return task_id


def run_video_job(task_id):
    """Run an existing task through all production stages."""

    task_dir = TASKS_DIR / task_id
    manifest_path = task_dir / "manifest.json"

    stage = "starting"

    try:
        manifest = json.loads(
            manifest_path.read_text(encoding="utf-8")
        )

        script = manifest["script"]
        keywords = manifest["keywords"]

        # Stage 1: Collect footage
        stage = "materials"

        update_job_status(
            task_id,
            state="processing",
            stage=stage,
        )

        downloaded = []
        used_ids = set()

        for keyword in keywords[:5]:
            if len(downloaded) >= 2:
                break

            keyword = str(keyword).strip()[:100]

            if not keyword:
                continue

            try:
                videos = search_videos(
                    keyword,
                    per_page=3,
                )

                for video in videos:
                    video_id = video.get("id")

                    if video_id in used_ids:
                        continue

                    if not video.get("videos", {}).get(
                        "small", {}
                    ).get("url"):
                        continue

                    try:
                        path = download_video(
                            video,
                            quality="small",
                        )
                    except MaterialError:
                        logger.warning(
                            "Could not download footage %s",
                            video_id,
                        )
                        continue

                    used_ids.add(video_id)

                    downloaded.append({
                        "keyword": keyword,
                        "video_id": video_id,
                        "source": video.get("pageURL"),
                        "file": str(path),
                    })

                    break

            except MaterialError:
                logger.warning(
                    "Pixabay search failed for a keyword"
                )

        if not downloaded:
            raise RuntimeError(
                "No video footage could be downloaded."
            )

        manifest["videos"] = downloaded

        manifest_path.write_text(
            json.dumps(manifest, indent=2),
            encoding="utf-8",
        )

        # Stage 2: Generate narration
        stage = "voiceover"

        update_job_status(
            task_id,
            stage=stage,
        )

        audio_path = task_dir / "audio.mp3"

        voice_result = generate_voiceover_sync(
            script=script,
            output_path=audio_path,
            voice=select_voice_for_script(script),
        )

        # Stage 3: Generate subtitles
        stage = "subtitles"

        update_job_status(
            task_id,
            stage=stage,
        )

        subtitle_path = task_dir / "subtitle.srt"

        generate_subtitles(
            metadata_path=voice_result["metadata_path"],
            output_path=subtitle_path,
        )

        # Stage 4: Render MP4
        stage = "rendering"

        update_job_status(
            task_id,
            stage=stage,
        )

        output_path = task_dir / "final_video.mp4"

        result = render_video(
            video_paths=[
                item["file"]
                for item in downloaded
            ],
            audio_path=audio_path,
            subtitle_path=subtitle_path,
            output_path=output_path,
            width=540,
            height=960,
            fps=20,
        )
        render_start = time.perf_counter()

        render_seconds = round(
            time.perf_counter() - render_start, 2
)

        
        
        print(f"FFmpeg rendering time: {render_seconds} seconds")

        update_job_status(
            task_id,
            state="completed",
            stage="finished",
            output=result,
            error=None,
        )

        logger.info(
            "Macy AI video task completed: %s",
            task_id,
        )

    
    except Exception as error:
        logger.exception(
            "Macy AI task %s failed during %s",
            task_id,
            stage,
        )

        error_type = type(error).__name__

        update_job_status(
            task_id,
            state="failed",
            stage=stage,
            error=(
                f"Task failed during {stage}. "
                f"Error type: {error_type}. "
                "Check FastAPI logs for details."
            ),
        )



def select_voice_for_script(script: str) -> str:
    """Choose a voice based on the script's writing system."""

    letters = [char for char in script if char.isalpha()]

    if not letters:
        return "en-AU-NatashaNeural"

    hindi_letters = sum(
        "\u0900" <= char <= "\u097F"
        for char in letters
    )

    hindi_ratio = hindi_letters / len(letters)

    if hindi_ratio >= 0.30:
        return "hi-IN-SwaraNeural"

    return "en-AU-NatashaNeural"
